# 第7章 数据存储与持久化

## 7.1 存储架构概述

### 7.1.1 为什么需要双存储

ScholarFlow采用**SQLite + JSON的双存储架构**，这是经过深思熟虑的设计决策：

**挑战**:
- LangGraph需要可靠的状态持久化支持断点续传
- 需要快速查询任务状态（避免数据库查询开销）
- 工作流状态复杂（30+字段），关系型数据库表设计困难
- 需要人类可读的状态用于调试

**解决方案**:
- **SQLite**: LangGraph原生的checkpoint存储，ACID事务保证
- **JSON**: 快速访问层，人类可读，易于调试

**优势**:

- ✅ 版本控制支持

### 7.2.2 SqliteSaver实现

**文件位置**: `app/storage/checkpointer.py`

```python
from langgraph.checkpoint.sqlite import SqliteSaver
import aiosqlite


class WorkflowCheckpointer:
    """工作流检查点管理器"""

    def __init__(self, db_path: str = "data/workflows/checkpoints.db"):
        self.db_path = db_path
        self.saver = SqliteSaver.from_conn_string(f"file:{db_path}?mode=rwc")

    async def save_state(self, task_id: str, state: dict, node_name: str):
        """保存状态到检查点"""
        config = {"configurable": {"thread_id": task_id}}

        # LangGraph 自动调用此方法
        # 我们在这里添加额外的逻辑

        # 记录执行日志
        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "node": node_name,
            "status": state.get("status"),
            "progress": state.get("progress_percentage")
        }
        state.setdefault("execution_log", []).append(log_entry)

        # 保存到 JSON（辅助存储）
        await self.save_to_json(task_id, state)

    async def load_state(self, task_id: str) -> Optional[dict]:
        """从检查点加载状态"""
        try:
            # 从 SQLIte 加载
            config = {"configurable": {"thread_id": task_id}}
            saved = await self.saver.aget(config)

            if saved:
                return saved.get("values", {})

            return None
        except Exception as e:
            logger.error(f"加载状态失败: {e}")
            return None

    async def save_to_json(self, task_id: str, state: dict):
        """保存状态到 JSON 文件"""
        file_path = Path(f"data/workflows/intermediate/{task_id}/{task_id}.json")
        file_path.parent.mkdir(parents=True, exist_ok=True)

        async with aiofiles.open(file_path, 'w', encoding='utf-8') as f:
            await f.write(json.dumps(state, indent=2, ensure_ascii=False))

    @property
    def sqlite_saver(self) -> SqliteSaver:
        """获取 SQLite saver"""
        return self.saver
```

**读取策略**:
```python
async def load_state(task_id: str) -> Optional[dict]:
    """从存储加载状态"""
    # 1. 优先从 SQLite 加载（最完整）
    state = await checkpointer.load_state(task_id)

    if state:
        return state

    # 2. 备选从 JSON 加载
    state = await load_from_json(task_id)

    if state:
        logger.info(f"从JSON恢复状态: {task_id}")
        return state

    # 3. 都没有，返回 None
    logger.warning(f"状态不存在: {task_id}")
    return None
```

### 7.4.2 数据一致性

**同步策略**:

```python
async def sync_stores(task_id: str):
    """同步两个存储的数据"""

    sqlite_state = await checkpointer.load_state(task_id)
```

**冲突解决**:
```python
async def resolve_conflict(task_id: str):
    """解决数据冲突"""
    sqlite_state = await checkpointer.load_state(task_id)
    json_state = await load_from_json(task_id)

    if not sqlite_state or not json_state:
        return

    # 比较时间戳
    sqlite_time = sqlite_state.get("updated_at", "")
    json_time = json_state.get("updated_at", "")

    if sqlite_time > json_time:
        # SQLite 更新，以 SQLite 为准
        await save_to_json(task_id, sqlite_state)
        logger.info(f"冲突解决: 使用 SQLite 数据")
    else:
        # JSON 更新，以 JSON 为准
        # 注意：这种情况很少见
        logger.warning(f"发现数据不一致: JSON 更新于 {json_time}")
```

## 7.5 任务管理

### 7.5.1 列出任务

**按状态筛选**:
```python
async def list_tasks(status: Optional[str] = None, limit: int = 100) -> List[dict]:
    """列出所有任务"""
    task_dir = Path("data/workflows/intermediate")

    if not task_dir.exists():
        return []

    tasks = []

    # 遍历所有任务目录
    for task_dir in task_dir.iterdir():
        if not task_dir.is_dir():
            continue

        json_file = task_dir / f"{task_dir.name}.json"

        if json_file.exists():
            try:
                async with aiofiles.open(json_file, 'r') as f:
                    content = await f.read()
                    state = json.loads(content)

                # 状态筛选
                if status is None or state.get("status") == status:
                    tasks.append({
                        "task_id": state["task_id"],
                        "status": state["status"],
                        "progress": state.get("progress_percentage", 0),
                        "created_at": state.get("created_at"),
                        "updated_at": state.get("updated_at"),
                        "pdf_path": state.get("pdf_path")
                    })

                    if len(tasks) >= limit:
                        break

            except Exception as e:
                logger.error(f"读取任务状态失败: {json_file}, {e}")

    # 按更新时间排序
    tasks.sort(key=lambda x: x.get("updated_at", ""), reverse=True)

    return tasks
```

### 7.5.2 任务统计

```python
async def get_task_statistics() -> dict:
    """获取任务统计信息"""
    all_tasks = await list_tasks()

    stats = {
        "total": len(all_tasks),
        "by_status": {},
        "avg_execution_time": 0,
        "total_execution_time": 0
    }

    # 按状态统计
    for task in all_tasks:
        status = task["status"]
        stats["by_status"][status] = stats["by_status"].get(status, 0) + 1

    # 计算执行时间
    execution_times = []
    for task in all_tasks:
        # 读取完整状态获取执行时间
        state = await load_from_json(task["task_id"])
        if state:
            exec_time = state.get("execution_time", 0)
            if exec_time > 0:
                execution_times.append(exec_time)

    if execution_times:
        stats["total_execution_time"] = sum(execution_times)
        stats["avg_execution_time"] = sum(execution_times) / len(execution_times)

    return stats
```

```python
async def backup_task(task_id: str):
    """备份任务数据"""
    backup_dir = Path(f"data/backups/{task_id}_{datetime.now().strftime('%Y%m%d_%H%M%S')}")

    # 1. 复制中间文件

            checkpoint_data.append(row)

    # 保存检查点数据
    if checkpoint_data:
        checkpoint_file = backup_dir / "checkpoints.json"
        async with aiofiles.open(checkpoint_file, 'w') as f:
            await f.write(json.dumps(checkpoint_data, indent=2))

    logger.info(f"任务已备份: {backup_dir}")
```

```python
async def restore_task(task_id: str, backup_path: str):
    """从备份恢复任务"""
    backup_dir = Path(backup_path)

    # 1. 恢复中间文件
    intermediate_backup = backup_dir / "intermediate"
    if intermediate_backup.exists():
        target_dir = Path(f"data/workflows/intermediate/{task_id}")
        if target_dir.exists():
            shutil.rmtree(target_dir)
        shutil.copytree(intermediate_backup, target_dir)

    # 2. 恢复检查点
    checkpoint_file = backup_dir / "checkpoints.json"
    if checkpoint_file.exists():
        async with aiofiles.open(checkpoint_file, 'r') as f:
            checkpoint_data = json.loads(await f.read())

        async with aiosqlite.connect(checkpointer.db_path) as db:
            await db.execute("DELETE FROM checkpoint WHERE thread_id = ?", (task_id,))
            for checkpoint in checkpoint_data:
                await db.execute(
                    "INSERT INTO checkpoint VALUES (?, ?, ?, ?, ?, ?, ?)",
                    checkpoint
                )
            await db.commit()

    logger.info(f"任务已恢复: {task_id}")
```

```python
async def cleanup_old_tasks(days: int = 30):
    """清理过期任务"""
    cutoff_time = datetime.now() - timedelta(days=days)

    all_tasks = await list_tasks()

    deleted_count = 0

    for task in all_tasks:
```

## 7.9 监控与告警

### 7.9.1 存储监控

```python
async def get_storage_stats() -> dict:
    """获取存储统计信息"""
    stats = {
        "sqlite": {},
        "json": {},
        "total_tasks": 0
    }

    # SQLite 统计
    async with aiosqlite.connect(checkpointer.db_path) as db:
        cursor = await db.execute("SELECT COUNT(*) FROM checkpoint")
        row = await cursor.fetchone()
        stats["sqlite"]["checkpoint_count"] = row[0]

        # 数据库大小
        db_path = Path(checkpointer.db_path)
        stats["sqlite"]["size_mb"] = db_path.stat().st_size / 1024 / 1024

    # JSON 统计
    task_dir = Path("data/workflows/intermediate")
    if task_dir.exists():
        task_count = len([d for d in task_dir.iterdir() if d.is_dir()])
        stats["json"]["task_count"] = task_count
        stats["total_tasks"] = task_count

        # 计算总大小
        total_size = sum(
            f.stat().st_size
            for f in task_dir.rglob("*")
            if f.is_file()
        )
        stats["json"]["total_size_mb"] = total_size / 1024 / 1024

    return stats
```

```python
        alerts.append({
            "type": "critical",
            "message": f"磁盘空间不足: {disk_usage.free / 1024 / 1024 / 1024:.2f}GB 可用"
        })

    return alerts
```

## 7.10 最佳实践

### 7.10.1 数据设计原则

1. **结构化存储**: SQLite存储结构化数据，JSON存储半结构化数据
2. **最小化冗余**: 避免数据重复存储
3. **版本控制**: 记录数据版本便于迁移
4. **压缩归档**: 定期压缩和归档旧数据
5. **备份策略**: 定期备份重要数据

### 7.10.2 性能优化建议

1. **批量操作**: 使用批量读写减少I/O次数
2. **索引优化**: 为常用查询字段建立索引
3. **连接池**: 使用连接池管理数据库连接
4. **缓存**: 缓存频繁访问的数据
5. **压缩**: 对大文件进行压缩存储

### 7.10.3 数据安全

1. **加密敏感数据**: 对API密钥等敏感信息加密
2. **访问控制**: 限制文件访问权限
3. **审计日志**: 记录所有数据访问操作
4. **定期清理**: 删除过期和不必要的数据
5. **备份验证**: 定期验证备份完整性
